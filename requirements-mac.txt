#   conda create -n aoi-daemon python=3.13 -y
#   conda activate aoi-daemon
#   python -m pip install -r requirements-mac.txt
#   python -m pip freeze > requirements-mac.lock.txt   # 完整快照，含间接依赖
#   python -c "import PySide6, live2d.v2, av; from l2d import LAppModel; print('ok')"

#   .venv/bin/pip install -r requirements-mac.txt
